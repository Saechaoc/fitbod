struct TodayView: View {
  @Environment(\.modelContext) private var ctx

  // From the existing RootView.TodayView (preserve verbatim)
  @Query(filter: #Predicate<Block> { $0.isActive == true })
  private var activeBlocks: [Block]
  @Query private var settingsList: [UserSettings]
  // NEW for BRU:
  @Query(filter: #Predicate<Session> { $0.completedAt == nil })
  private var activeSessions: [Session]
  @Query(sort: \Session.startedAt, order: .reverse)
  private var allSessions: [Session]
  // Today's scheduled routine (from active block / settings — preserved logic from existing TodayView):
  // (lookup logic carries forward from current RootView.TodayView)

  @State private var navigationPath = NavigationPath()
  @State private var pendingBlockBuilder: PendingBlockBuilder? = nil
  @State private var pendingBlockReview: Block? = nil
  // Phase 5 plan 04 — deload advisory state.
  @State private var currentAdvisory: DeloadAdvisory? = nil
  @State private var presentingSignalSheet: Bool = false
  // Phase 5 plan 06 — weekly recap state.
  @State private var presentingWeeklyRecap: Bool = false

  var body: some View {
    NavigationStack(path: $navigationPath) {
      ScrollView {
        VStack(spacing: 0) {
          // === DEFENSIVE TOP — Phase 5 banners (preserved from existing TodayView) ===
          if let active = activeBlocks.first,
             let phase = PeriodizationEngine.phase(for: active, on: .now),
             phase.kind == .deload {
            DeloadWeekBanner(currentPhaseKind: phase.kind)
          }
          if let advisory = currentAdvisory {
            DeloadAdvisoryBanner(advisory: advisory,
                                 onDismiss: { dismissAdvisory() },
                                 onTapBody: { presentingSignalSheet = true })
          }

          // === BRU LAYOUT (this is what's NEW) ===
          TodayStatStrip(streak: computedStreak, totalSessions: allSessions.count,
                         weekIndex: currentWeekIndex, totalWeeks: currentTotalWeeks,
                         trend: currentTrend)
          if let block = activeBlocks.first {
            BlockContextBand(block: block)
          }
          if let active = activeSessions.first {
            ResumeWorkoutSlab(session: active,
              onResume: { navigationPath.append(SessionRoute.logger(active)) },
              onDiscard: { ctx.delete(active); try? ctx.save() })
          } else if let routine = todaysRoutine {
            BlackSlabCTA(routine: routine, dayIndex: dayIndexInBlock, totalDays: totalDaysInBlock,
              onStart: { startWorkout(routine) })
          } else if activeBlocks.isEmpty {
            StartBlockCTA(onStart: { pendingBlockBuilder = .template(BlockTemplates.blank) })
          } else {
            // Empty state per UI-SPEC line 466
            VStack(spacing: 12) {
              Text("No workout scheduled").font(BRUFont.displayLarge).foregroundStyle(.bruInk)
              Text("Build a routine or pick one from your library.")
                .font(BRUFont.bodyDefault).foregroundStyle(.bruMuted).multilineTextAlignment(.center)
              Button { /* navigate to RoutinesListView's +NEW */ } label: {
                Text("+ NEW ROUTINE").font(BRUFont.labelSmall).tracking(1.2)
                  .textCase(.uppercase).foregroundStyle(.bruSignal)
              }
            }.padding(.vertical, 32).padding(.horizontal, 16)
          }
          LiftListSection(routine: todaysRoutine)  // empty when nil

          // === DEFENSIVE BOTTOM — Phase 5 FatigueSurfaceView (preserved) ===
          FatigueSurfaceView(isAdvisoryActive: currentAdvisory != nil,
                             navigationPath: $navigationPath)
        }
      }
      .background(Color.bruPaper.ignoresSafeArea())
      .toolbar { ToolbarItem(placement: .principal) { EmptyView() } }  // suppress default chrome
      .navigationBarHidden(true)  // BRUHeaderBar replaces it
      .safeAreaInset(edge: .top) {
        BRUHeaderBar(
          left: nil,
          center: "Fitbod / Today",
          right: HeaderAction(formattedDate, action: { /* no-op or open date picker */ })
        )
      }
      // (existing .navigationDestination + .sheet + .task wiring preserved)
    }
  }
}
