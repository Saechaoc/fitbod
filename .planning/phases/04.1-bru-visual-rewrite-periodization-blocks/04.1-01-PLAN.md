  HeaderSlot enum: .back / .cancel / .action(String, () -> Void) / .text(String) / .none
  rendering: 56pt min height, bruPaper background, 2pt bruInk bottom border, left/center/right HStack; labelSmall on slots; tracking 1.5; uppercase on action+text slots

BRUSectionHead:
  init(_ title: String, right: String? = nil, rightAction: (() -> Void)? = nil)
  rendering: HStack — title labelSmall uppercase tracked 1.2 in bruInk + Spacer + optional right labelSmall uppercase tracked 1.2 in bruSignal (tappable when rightAction provided); padding horizontal 16, vertical 12

  init(_ kind: Kind = .primary, _ label: String, action: @escaping () -> Void)
  Kind enum: .primary (bruInk fill, bruPaper text), .signal (bruSignal fill, bruPaper text), .ghost (clear fill, bruInk text, 2pt bruInk border)
  rendering: full-width, 48pt min height, sharp 0-radius corners, labelSmall uppercase tracked 1.2

BRUDivider:
  init(_ weight: Weight)
  Weight enum: .hair (1pt bruHair fill), .ink (2pt bruInk fill)
  rendering: Rectangle().fill(...).frame(height: 1 or 2)

BRUDashedSlot:
  init(_ caption: String, action: (() -> Void)? = nil)
  rendering: dashed bruHair border (StrokeStyle(lineWidth: 1, dash: [4, 4])) on transparent fill, 32-48pt min height, caption in bodyDefault muted

BRUPillButton:
  init(_ label: String, fill: Color = .bruSignal, action: @escaping () -> Void)
  rendering: rectangular signal-fill (NOT Capsule); 12pt horizontal / 8pt vertical padding; labelSmall uppercase tracked 1.2 in bruPaper

BRUDisplay:
  init(_ value: String, size: Size = .large, color: Color = .bruInk)
  Size enum: .hero (BRUFont.displayHero 56pt), .large (BRUFont.displayLarge 32pt), .medium (BRUFont.displayMedium 18pt)
  rendering: Text(value).font(...).foregroundStyle(color)

BRUReadout:
  init(label: String, value: String, valueColor: Color = .bruInk, size: BRUDisplay.Size = .medium)
  rendering: VStack(spacing: 4, alignment: .leading) — labelMicro uppercase tracked 1.0 muted caption above BRUDisplay value
