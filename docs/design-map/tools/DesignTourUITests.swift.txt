//
//  DesignTourUITests.swift
//  Design-map screenshot tour for docs/design-map/.
//
//  This file is not part of the app's test suite. tools/tour_run.sh copies
//  it into fitbodUITests/ of a private worktree copy of the project on
//  every run, so the shared helpers there (UITestNavigation and friends)
//  compile with it and nothing in the repository's targets changes.
//
//  The tour only navigates and captures. It asserts only that each screen
//  appeared, and it changes no app behavior.
//
//  Environment (passed by xcodebuild through the TEST_RUNNER_ prefix):
//    DESIGN_TEXTSIZE   default | ax3
//    DESIGN_APPEARANCE light | dark   (names files; the run sets the phone)
//    DESIGN_TREES      1 attaches app.debugDescription beside each shot
//
//  Screenshot names: <phone>-<size>-<appearance>-<screen>[--<state>],
//  where phone is 17e (390pt) or 17promax (440pt) and screen is the
//  screens/ file stem.
//

import XCTest

// MARK: - Helpers

@MainActor
enum DesignTour {
    private static var env: [String: String] { ProcessInfo.processInfo.environment }

    static var textSize: UITestNavigation.TextSize { env["DESIGN_TEXTSIZE"] == "ax3" ? .ax3 : .default }
    static var isAX3: Bool { textSize == .ax3 }
    static var appearance: String { env["DESIGN_APPEARANCE"] == "dark" ? "dark" : "light" }
    static var trees: Bool { env["DESIGN_TREES"] == "1" }

    private static var cachedPhone: String?

    /// "17e" on the 390pt iPhone 17e, "17promax" on the 440pt iPhone 17 Pro
    /// Max, else the width in points.
    static var phone: String {
        if let cachedPhone { return cachedPhone }
        let width = Int(XCUIScreen.main.screenshot().image.size.width.rounded())
        let name = switch width {
        case 390: "17e"
        case 440: "17promax"
        default: "\(width)pt"
        }
        cachedPhone = name
        return name
    }

    /// Launches on a fresh store with `arguments` (seed flags) at the run's
    /// text size, then waits up to 30 s for `ready`, an element the seed
    /// creates, because seeds can finish after the tab bar appears.
    @discardableResult
    static func launch(
        _ arguments: [String] = [],
        ready: ((XCUIApplication) -> XCUIElement)? = nil,
        file: StaticString = #filePath, line: UInt = #line
    ) -> XCUIApplication {
        let app = UITestNavigation.launch([UITestNavigation.resetFlag] + arguments, textSize: textSize, file: file, line: line)
        if let ready {
            let element = ready(app)
            XCTAssertTrue(element.waitForExistence(timeout: 30), "The seed never showed \(element)", file: file, line: line)
        }
        assertAppearance(file: file, line: line)
        return app
    }

    /// Relaunches without the reset flag, the way a user reopens the app.
    static func relaunch(_ app: XCUIApplication, file: StaticString = #filePath, line: UInt = #line) {
        UITestNavigation.relaunch(app, textSize: textSize, file: file, line: line)
    }

    private static var checkedAppearance = false

    /// Fails once per run when the phone's appearance is not the one the
    /// run names, so a light run never files dark screenshots.
    private static func assertAppearance(file: StaticString, line: UInt) {
        guard !checkedAppearance else { return }
        checkedAppearance = true
        let screenshot = XCUIScreen.main.screenshot()
        guard let pixel = UITestNavigation.color(at: CGPoint(x: 6, y: 6), in: screenshot) else { return }
        let isDark = (pixel.red + pixel.green + pixel.blue) / 3 < 128
        XCTAssertEqual(isDark ? "dark" : "light", appearance,
                       "The phone shows \(pixel) at the top, which does not match DESIGN_APPEARANCE", file: file, line: line)
    }

    /// Attaches a full-screen screenshot named
    /// `<phone>-<size>-<appearance>-<screen>` and, with DESIGN_TREES=1, the
    /// accessibility tree as `<same>.tree`. It waits until `still` stops
    /// moving, or a short settle when there is none.
    static func shot(_ app: XCUIApplication, _ screen: String, still: XCUIElement? = nil) {
        if let still, still.exists {
            UITestNavigation.waitUntilStill(still)
        } else {
            Thread.sleep(forTimeInterval: 0.8)
        }
        let name = [phone, textSize.rawValue, appearance, screen].joined(separator: "-")
        XCTContext.runActivity(named: "Shot \(name)") { activity in
            let image = XCTAttachment(screenshot: XCUIScreen.main.screenshot())
            image.name = name
            image.lifetime = .keepAlways
            activity.add(image)
            if trees {
                let tree = XCTAttachment(string: app.debugDescription)
                tree.name = name + ".tree"
                tree.lifetime = .keepAlways
                activity.add(tree)
            }
        }
    }

    /// Runs one capture step as a named activity. A thrown error is
    /// recorded and the walk goes on, as failed assertions do.
    static func step(_ name: String, _ body: () throws -> Void) {
        XCTContext.runActivity(named: name) { _ in
            do {
                try body()
            } catch {
                XCTFail("\(name) threw \(error)")
            }
        }
    }

    /// The first tappable element in `query` whose label is `label` (any
    /// case, or followed by a comma), after up to 10 s.
    @discardableResult
    static func tap(_ query: XCUIElementQuery, _ label: String, exact: Bool = false,
                    file: StaticString = #filePath, line: UInt = #line) -> XCUIElement {
        let element = UITestNavigation.hittable(query, label, exactLabel: exact, file: file, line: line)
        element.tap()
        return element
    }

    /// True once `element` exists, within `timeout` seconds; records a
    /// failure that names `what` otherwise.
    @discardableResult
    static func expect(_ element: XCUIElement, _ what: String, timeout: TimeInterval = 10,
                       file: StaticString = #filePath, line: UInt = #line) -> Bool {
        let found = element.waitForExistence(timeout: timeout)
        XCTAssertTrue(found, "\(what) never appeared", file: file, line: line)
        return found
    }

    /// A static text whose label is `label` in any case.
    static func text(_ app: XCUIApplication, _ label: String) -> XCUIElement {
        app.staticTexts.labeled(label)
    }

    /// A static text whose label contains `fragment`.
    static func text(_ app: XCUIApplication, containing fragment: String) -> XCUIElement {
        app.staticTexts.matching(NSPredicate(format: "label CONTAINS[c] %@", fragment)).firstMatch
    }

    /// Any element whose label contains `fragment`.
    static func any(_ app: XCUIApplication, containing fragment: String) -> XCUIElement {
        app.descendants(matching: .any).matching(NSPredicate(format: "label CONTAINS[c] %@", fragment)).firstMatch
    }

    /// Waits for an alert, or a confirmation dialog showing `title`, and
    /// returns whether one came up. iOS 26 draws a confirmation dialog as a
    /// popover over a PopoverDismissRegion.
    @discardableResult
    static func waitForDialog(_ app: XCUIApplication, title: String? = nil, timeout: TimeInterval = 10,
                              file: StaticString = #filePath, line: UInt = #line) -> Bool {
        let deadline = Date.now.addingTimeInterval(timeout)
        while Date.now < deadline {
            if app.alerts.firstMatch.exists { return true }
            if let title, app.staticTexts.labeled(title).exists { return true }
            if title == nil, app.otherElements["PopoverDismissRegion"].exists { return true }
            Thread.sleep(forTimeInterval: 0.25)
        }
        XCTFail("No dialog\(title.map { " titled \($0)" } ?? "") came up", file: file, line: line)
        return false
    }

    /// Taps the visible alert or dialog button labeled `label`.
    static func tapDialogButton(_ app: XCUIApplication, _ label: String,
                                file: StaticString = #filePath, line: UInt = #line) {
        let alert = app.alerts.firstMatch
        if alert.exists {
            let button = alert.buttons.labeled(label)
            if button.waitForExistence(timeout: 5) {
                button.tap()
                return
            }
        }
        UITestNavigation.hittable(app.buttons, label, exactLabel: true, file: file, line: line).tap()
    }

    /// Closes a confirmation dialog or popover without choosing, the way a
    /// user taps outside it. iOS 26 hides a cancel-role button.
    static func dismissDialog(_ app: XCUIApplication) {
        let outside = app.otherElements["PopoverDismissRegion"]
        if outside.waitForExistence(timeout: 3) {
            outside.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.1)).tap()
        } else {
            app.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.08)).tap()
        }
        Thread.sleep(forTimeInterval: 0.6)
    }

    /// Swipes the top sheet down from its top edge, the way a user drags
    /// its grabber, and waits for it to settle.
    static func closeSheet(_ app: XCUIApplication, file: StaticString = #filePath, line: UInt = #line) {
        let deadline = Date.now.addingTimeInterval(5)
        var sheet = topSheet(in: app)
        while sheet == nil && Date.now < deadline {
            Thread.sleep(forTimeInterval: 0.25)
            sheet = topSheet(in: app)
        }
        guard let sheet else { return XCTFail("No sheet is on screen to swipe down", file: file, line: line) }
        let topEdge = sheet.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0)).withOffset(CGVector(dx: 0, dy: 6))
        let bottom = app.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.98))
        topEdge.press(forDuration: 0.05, thenDragTo: bottom, withVelocity: .fast, thenHoldForDuration: 0)
        Thread.sleep(forTimeInterval: 0.8)
    }

    /// The sheet on top, or nil. This is UITestNavigation.topSheet's rule
    /// with room for an iOS 26 half-height sheet, which floats 8 pt above
    /// the bottom of the screen, so that helper finds no sheet.
    static func topSheet(in app: XCUIApplication) -> XCUIElement? {
        let window = app.windows.firstMatch
        guard let layer = window.children(matching: .other).allElementsBoundByIndex.last else { return nil }
        let bottom = window.frame.maxY
        return layer.children(matching: .other).allElementsBoundByIndex.first {
            $0.frame.minY > 0.5 && abs($0.frame.maxY - bottom) < 12
        }
    }

    /// Scrolls until `element` is visible and tappable. UITestNavigation's
    /// reveal only scrolls down the page; this one also scrolls back up to
    /// an element above the screen. A row under a header bar drawn as a top
    /// safe-area inset still counts as hittable; use tapBelow for those.
    static func reveal(_ element: XCUIElement, in app: XCUIApplication, maxSwipes: Int = 12,
                       file: StaticString = #filePath, line: UInt = #line) {
        for _ in 0..<maxSwipes {
            if element.exists && element.isHittable { return }
            let target: XCUIElement = app.scrollViews.allElementsBoundByIndex.last(where: { $0.isHittable }) ?? app
            let frame = element.exists ? element.frame : .zero
            if !frame.isEmpty && frame.midY < app.frame.midY {
                target.swipeDown(velocity: .slow)
            } else {
                target.swipeUp(velocity: .slow)
            }
        }
        XCTAssertTrue(element.exists && element.isHittable, "Expected a visible control: \(element)", file: file, line: line)
    }

    /// Swipes the screen's main scroll view up `times` times, slowly, for
    /// screenshots of a long screen's lower part.
    static func scrollDown(_ app: XCUIApplication, times: Int = 1) {
        for _ in 0..<times {
            if let scroll = app.scrollViews.allElementsBoundByIndex.last(where: { $0.isHittable }) {
                scroll.swipeUp(velocity: .slow)
            } else {
                app.swipeUp(velocity: .slow)
            }
        }
        Thread.sleep(forTimeInterval: 0.6)
    }

    /// The system back gesture on a pushed screen.
    static func back(_ app: XCUIApplication) {
        UITestNavigation.swipeBack(in: app)
        Thread.sleep(forTimeInterval: 0.8)
    }
}

// MARK: - Area helpers

@MainActor
enum TourTodaySettings {
    static func recapTitle(in app: XCUIApplication) -> XCUIElement {
        app.staticTexts.matching(NSPredicate(format: "label ENDSWITH[c] %@", " · recap")).firstMatch
    }

    static func reviewTitle(in app: XCUIApplication) -> XCUIElement {
        app.staticTexts.labeled("Block review")
    }

    /// Waits on Progress until the seed has finished `count` workouts, then
    /// opens Today, as WeeklyRecapUITests does: Today's first visit decides
    /// whether the recap shows.
    static func openToday(afterSeeding count: Int, in app: XCUIApplication) {
        UITestNavigation.selectTab("Progress", in: app)
        let seeded = XCTNSPredicateExpectation(
            predicate: NSPredicate(format: "label CONTAINS %@", "\(count) finished"),
            object: app.buttons["progress.history"]
        )
        XCTAssertEqual(XCTWaiter().wait(for: [seeded], timeout: 30), .completed,
                       "The seed must finish \(count) workouts before Today's first visit")
        UITestNavigation.selectTab("Today", in: app)
    }

    /// Closes the recap with its footer's Start button, such as "Start W44 ▸".
    static func closeRecap(in app: XCUIApplication) {
        let start = app.buttons.matching(NSPredicate(format: "label BEGINSWITH[c] %@", "Start W")).firstMatch
        XCTAssertTrue(start.waitForExistence(timeout: 5), "The recap must offer its Start button")
        start.tap()
        _ = recapTitle(in: app).waitForNonExistence(timeout: 5)
    }

    /// Opens the demo workout and checks off set 1 of the first exercise,
    /// as SkipWarmupsUITests does. Leaves the logger open.
    static func logFirstSet(in app: XCUIApplication) {
        TourLogger.openWorkout(in: app)
        // At AX3 the set rows load only as they scroll in.
        TourLogger.logSet("1.S1", reps: "6", in: app)
        // The first rest asks to send notifications, once per install. Shoot
        // it here, before XCUITest's interruption handler answers it unseen.
        let prompt = XCUIApplication(bundleIdentifier: "com.apple.springboard").alerts.firstMatch
        if prompt.waitForExistence(timeout: 5) {
            DesignTour.shot(app, "notification-permission")
            prompt.buttons["Allow"].tap()
            _ = prompt.waitForNonExistence(timeout: 5)
        }
    }

    /// Scrolls Today until both heatmap figures sit above the tab bar, as
    /// HeatmapHitAreaUITests does.
    static func revealHeatmap(in app: XCUIApplication) {
        let back = app.otherElements.matching(NSPredicate(format: "label == %@", "Back")).firstMatch
        UITestNavigation.reveal(back, in: app)
        let tabBarTop = UITestNavigation.tab("Today", in: app).frame.minY
        for _ in 0..<4 where back.frame.maxY + 14 > tabBarTop {
            app.swipeUp(velocity: .slow)
        }
    }

    /// Scrolls Plate Inventory until `element` sits wholly above its
    /// Done ▸ footer, as PlateInventorySwipeUITests does.
    static func revealAboveFooter(_ element: XCUIElement, in app: XCUIApplication) {
        let footerTop = app.buttons.labeled("Done ▸").frame.minY - 2
        for _ in 0..<6 {
            if element.exists && element.isHittable && element.frame.maxY <= footerTop { return }
            app.swipeUp(velocity: .slow)
        }
        XCTAssertTrue(element.exists && element.isHittable, "\(element) must sit above the footer bar")
    }

    static func shareSheet(in app: XCUIApplication) -> XCUIElement {
        app.otherElements.matching(identifier: "ActivityListView").firstMatch
    }

    static func file(_ name: String, in app: XCUIApplication) -> XCUIElement {
        app.descendants(matching: .any).matching(NSPredicate(format: "label BEGINSWITH %@", name)).firstMatch
    }

    static func waitForAny(_ elements: [XCUIElement], timeout: TimeInterval) -> Bool {
        let deadline = Date.now.addingTimeInterval(timeout)
        while Date.now < deadline {
            if elements.contains(where: \.exists) { return true }
            Thread.sleep(forTimeInterval: 0.25)
        }
        return false
    }

    /// Backup CREATE, then Save to Files under a unique name, as
    /// SettingsExportUITests does. Returns the name.
    static func saveBackupToFiles(in app: XCUIApplication) -> String {
        let create = app.buttons.labeled("Create")
        XCTAssertTrue(create.waitForExistence(timeout: 10), "Missing backup CREATE")
        UITestNavigation.reveal(create, in: app)
        create.tap()
        XCTAssertTrue(shareSheet(in: app).waitForExistence(timeout: 5), "CREATE must open the share sheet")
        let saveToFiles = app.cells.labeled("Save to Files")
        XCTAssertTrue(saveToFiles.waitForExistence(timeout: 5), "The share sheet must offer Save to Files")
        saveToFiles.tap()

        let save = app.buttons.labeled("Save")
        let nameField = app.textFields["DOCPicker.filenameTextField"]
        XCTAssertTrue(save.waitForExistence(timeout: 15), "Files must offer Save")
        let name = "fitbod-backup-\(UUID().uuidString.prefix(8))"
        nameField.tap()
        let focused = XCTNSPredicateExpectation(predicate: NSPredicate(format: "hasKeyboardFocus == true"), object: nameField)
        XCTAssertEqual(XCTWaiter().wait(for: [focused], timeout: 5), .completed, "The name must take the keyboard")
        // Focus selects the whole name, so typing replaces it.
        nameField.typeText(name)
        save.tap()
        XCTAssertTrue(nameField.waitForNonExistence(timeout: 10), "Files must close after saving")
        return name
    }

    /// Picks `name` in the open Files picker, through Browse > On My iPhone
    /// when Recents does not list it.
    static func pickFile(_ name: String, in app: XCUIApplication) {
        let file = file(name, in: app)
        if !file.waitForExistence(timeout: 10) {
            let browse = app.buttons.labeled("Browse")
            XCTAssertTrue(browse.waitForExistence(timeout: 5), "The Files picker must offer Browse")
            browse.tap()
            let onMyPhone = app.descendants(matching: .any).matching(NSPredicate(format: "label ==[c] %@", "On My iPhone")).firstMatch
            XCTAssertTrue(onMyPhone.waitForExistence(timeout: 10), "Browse must list On My iPhone")
            onMyPhone.tap()
        }
        XCTAssertTrue(file.waitForExistence(timeout: 10), "The Files picker must list \(name)")
        file.tap()
    }
}

@MainActor
enum TourLogger {
    /// The demo, stalled and calibrating seeds' first exercise.
    static let bench = "Barbell Bench Press - Medium Grip"

    /// Opens today's workout from Today's resume slab.
    static func openWorkout(in app: XCUIApplication) {
        let resume = app.buttons["resumeSlab.resume"]
        XCTAssertTrue(resume.waitForExistence(timeout: 10), "Today must offer the workout")
        UITestNavigation.reveal(resume, in: app)
        resume.tap()
        XCTAssertTrue(app.otherElements["workout.header"].waitForExistence(timeout: 10), "The logger never opened")
    }

    static func element(_ app: XCUIApplication, beginning prefix: String) -> XCUIElement {
        app.descendants(matching: .any).matching(NSPredicate(format: "label BEGINSWITH %@", prefix)).firstMatch
    }

    static func restStatus(in app: XCUIApplication) -> XCUIElement {
        app.descendants(matching: .any).matching(identifier: "rest.status").firstMatch
    }

    /// Scrolls the logger down (or up) until `element` loads, since rows
    /// load as they scroll in, then into the span a tap reaches.
    static func revealLoaded(_ element: XCUIElement, in app: XCUIApplication, up: Bool = false) {
        // The demo workout runs about 11,000 pt at AX3, some 22 drags.
        for _ in 0..<30 where !element.exists {
            let (top, bottom) = UITestNavigation.loggerSpan(in: app)
            UITestNavigation.dragLogger(from: up ? top + 20 : bottom - 20, to: up ? bottom - 20 : top + 20, in: app)
        }
        UITestNavigation.revealInLogger(element, in: app)
    }

    /// Types reps into set `id`, such as "1.S1", closes the keyboard with
    /// its DONE, which covers the checkmark, and checks the set off.
    static func logSet(_ id: String, reps: String, in app: XCUIApplication) {
        let field = app.textFields["set.reps.\(id)"]
        revealLoaded(field, in: app)
        UITestNavigation.type(reps, into: field, replacing: false, pressingReturn: false)
        UITestNavigation.hittable(app.buttons, "Done", exactLabel: true).tap()
        _ = app.keyboards.firstMatch.waitForNonExistence(timeout: 5)
        let complete = app.buttons["set.complete.\(id)"]
        UITestNavigation.revealInLogger(complete, in: app)
        complete.tap()
        let number = id.split(separator: "S").last.map(String.init) ?? ""
        let logged = XCTNSPredicateExpectation(predicate: NSPredicate(format: "label == %@", "Set \(number) complete"), object: complete)
        XCTAssertEqual(XCTWaiter().wait(for: [logged], timeout: 5), .completed, "Set \(id) must be logged")
    }

    /// Long-presses the first exercise's name and picks `item` from its
    /// context menu: Swap Exercise…, Edit Pinned Note or Remove from Session.
    static func exerciseMenu(_ item: String, in app: XCUIApplication) {
        let name = app.staticTexts[bench].firstMatch
        revealLoaded(name, in: app, up: true)
        name.press(forDuration: 1.2)
        let button = app.buttons[item]
        XCTAssertTrue(button.waitForExistence(timeout: 5), "The exercise menu must offer \(item)")
        button.tap()
    }
}

@MainActor
enum TourBlocks {
    /// A Program block card, read as "READY block: Summer Block" and, for
    /// an active block, its week after a period.
    static func card(_ prefix: String, in app: XCUIApplication) -> XCUIElement {
        app.buttons.matching(NSPredicate(format: "label BEGINSWITH[c] %@", prefix)).firstMatch
    }

    /// The block overview's hero, one element read as "Ready block #2: …".
    static func hero(in app: XCUIApplication) -> XCUIElement {
        app.staticTexts.matching(NSPredicate(
            format: "label MATCHES[c] %@", "(active|ready|draft|completed) block .*"
        )).firstMatch
    }

    /// Opens a block's overview from its Program card, as
    /// BlockOverviewStatesUITests does.
    static func openBlock(_ prefix: String, in app: XCUIApplication) {
        UITestNavigation.selectTab("Program", in: app)
        let button = card(prefix, in: app)
        XCTAssertTrue(button.waitForExistence(timeout: 10), "Program must list \(prefix)")
        UITestNavigation.reveal(button, in: app)
        button.tap()
        XCTAssertTrue(hero(in: app).waitForExistence(timeout: 10), "\(prefix) must open the block overview")
    }

    /// Taps the dimmed screen beside + NEW's popover, which closes it, as
    /// ProgramHeaderAndSheetsAX3UITests does.
    static func closePopover(in app: XCUIApplication) {
        app.coordinate(withNormalizedOffset: CGVector(dx: 0.25, dy: 0.75)).tap()
        XCTAssertTrue(app.buttons["+ New block"].waitForNonExistence(timeout: 5), "A tap beside the popover must close it")
    }

    /// Program's + NEW, then + New block: the CREATE BLOCK sheet.
    static func openStartChoice(in app: XCUIApplication) {
        let new = app.buttons["+ NEW"]
        XCTAssertTrue(new.waitForExistence(timeout: 10), "Program has no + NEW")
        new.tap()
        UITestNavigation.hittable(app.buttons, "+ New block").tap()
        XCTAssertTrue(app.staticTexts["CREATE BLOCK"].waitForExistence(timeout: 5), "+ New block must open CREATE BLOCK")
    }

    /// Waits for New Block, until its CANCEL can be tapped, so the cover or
    /// sheet has finished sliding up.
    static func expectBuilder(in app: XCUIApplication, file: StaticString = #filePath, line: UInt = #line) {
        XCTAssertTrue(app.staticTexts.labeled("NEW BLOCK").waitForExistence(timeout: 10), "New Block never opened",
                      file: file, line: line)
        UITestNavigation.hittable(app.buttons, "Cancel", file: file, line: line)
    }

    /// CANCEL closes an untouched builder at once and asks first after an
    /// edit; this takes Discard when it asks.
    static func closeBuilder(in app: XCUIApplication, title: String = "NEW BLOCK") {
        UITestNavigation.hittable(app.buttons, "Cancel", exactLabel: true).tap()
        let discard = app.buttons.matching(NSPredicate(format: "label ==[c] %@", "Discard")).firstMatch
        if discard.waitForExistence(timeout: 2) {
            UITestNavigation.hittable(app.buttons, "Discard", exactLabel: true).tap()
        }
        XCTAssertTrue(app.staticTexts.labeled(title).waitForNonExistence(timeout: 5), "CANCEL must close \(title)")
    }

    /// Taps + ADD PHASE and returns the sheet's hint line.
    @discardableResult
    static func openAddPhase(in app: XCUIApplication) -> XCUIElement {
        UITestNavigation.revealAboveBottomBar(app.buttons, "+ ADD PHASE", in: app).tap()
        let prompt = app.staticTexts.labeled("Choose a phase type to add.")
        XCTAssertTrue(prompt.waitForExistence(timeout: 10), "+ ADD PHASE must open its sheet")
        return prompt
    }

    /// The phase card labeled `label`, such as "Phase 1: Accumulation".
    static func phaseCard(_ label: String, in app: XCUIApplication) -> XCUIElement {
        let card = app.descendants(matching: .any).matching(NSPredicate(format: "label == %@", label)).firstMatch
        XCTAssertTrue(card.waitForExistence(timeout: 10), "Missing \(label)")
        return card
    }

    /// Taps a phase card and returns its editor's Done.
    static func openPhaseEditor(_ label: String, in app: XCUIApplication) -> XCUIElement {
        let card = phaseCard(label, in: app)
        revealInBuilder(card, in: app)
        card.tap()
        let done = app.buttons.labeled("Done")
        XCTAssertTrue(done.waitForExistence(timeout: 5), "\(label) must open the phase editor")
        return done
    }

    /// Scrolls a block builder until `element` sits between its title and
    /// its bottom bar, with drags in the leading margin like
    /// UITestNavigation.revealAboveBottomBar's .dragInMargin. That helper
    /// measures from the NEW BLOCK title only, so it fails in EDIT BLOCK.
    /// A drag that starts beside a phase card does not scroll the page
    /// (T120), so each drag starts at a height no card covers. When cards
    /// cover every height, the caller's tap scrolls the element into view.
    static func revealInBuilder(_ element: XCUIElement, in app: XCUIApplication,
                                file: StaticString = #filePath, line: UInt = #line) {
        guard let title = ["NEW BLOCK", "EDIT BLOCK"].map({ app.staticTexts.labeled($0) }).first(where: { $0.exists }) else {
            return XCTFail("No block builder title is on screen", file: file, line: line)
        }
        let pageTop = title.frame.maxY + 20
        // The bar's 52 pt buttons sit 24 pt below its top edge.
        let barTop = UITestNavigation.hittable(app.buttons, "Cancel", file: file, line: line).frame.midY - 50
        let cards = app.buttons.matching(NSPredicate(format: "label BEGINSWITH %@", "Phase "))
        let origin = app.windows.firstMatch.coordinate(withNormalizedOffset: .zero)
        for _ in 0..<8 {
            let frame = element.frame
            if frame.minY >= pageTop && frame.maxY <= barTop { return }
            let up = frame.maxY > barTop
            let covered = cards.allElementsBoundByIndex.map { ($0.frame.minY - 8)...($0.frame.maxY + 8) }
            // Lowest free height for a drag up, highest for a drag down.
            let heights = Array(stride(from: barTop - 10, through: pageTop + 10, by: -10))
            guard let y = (up ? heights : heights.reversed()).first(where: { y in !covered.contains { $0.contains(y) } }) else {
                XCTContext.runActivity(named: "Phase cards cover the page; the tap scrolls \(element)") { _ in }
                return
            }
            let distance = up ? -(frame.maxY - barTop + 80) : pageTop - frame.minY + 80
            let start = origin.withOffset(CGVector(dx: 8, dy: y))
            let end = origin.withOffset(CGVector(dx: 8, dy: min(app.frame.maxY - 40, max(60, y + distance))))
            start.press(forDuration: 0.05, thenDragTo: end, withVelocity: .slow, thenHoldForDuration: 0.3)
            UITestNavigation.waitUntilStill(element)
        }
        XCTFail("\(element) must scroll between the builder's bars", file: file, line: line)
    }

    /// A slow, short drag reveals the card's Delete, as
    /// PhaseRowsVoiceOverUITests does; a fast full swipe deletes at once.
    /// Every card keeps a Delete behind it, so this takes the one on the
    /// card's row, not the first in the tree.
    static func revealDelete(on card: XCUIElement, in app: XCUIApplication) -> XCUIElement {
        let start = card.coordinate(withNormalizedOffset: CGVector(dx: 0.7, dy: 0.5))
        let end = card.coordinate(withNormalizedOffset: CGVector(dx: 0.4, dy: 0.5))
        start.press(forDuration: 0.05, thenDragTo: end, withVelocity: .slow, thenHoldForDuration: 0.1)
        let deletes = app.buttons.matching(NSPredicate(format: "label ==[c] %@", "Delete"))
        XCTAssertTrue(deletes.firstMatch.waitForExistence(timeout: 5), "The drag must reveal a Delete")
        let row = card.frame.midY
        let delete = deletes.allElementsBoundByIndex.first { abs($0.frame.midY - row) < 4 } ?? deletes.firstMatch
        UITestNavigation.waitUntilStill(delete)
        return delete
    }

    /// Opens New Block's routine picker with `control` and returns its title.
    static func openRoutinePicker(from control: String, in app: XCUIApplication) -> XCUIElement {
        let open = app.buttons.labeled(control)
        XCTAssertTrue(open.waitForExistence(timeout: 10), "New Block must offer \(control)")
        revealInBuilder(open, in: app)
        open.tap()
        let picker = app.staticTexts.labeled("Choose Routine")
        XCTAssertTrue(picker.waitForExistence(timeout: 5), "\(control) must open the routine picker")
        return picker
    }

    /// Opens the routine picker with `control` and picks `routine`.
    static func pickRoutine(_ routine: String, from control: String, in app: XCUIApplication) {
        let picker = openRoutinePicker(from: control, in: app)
        UITestNavigation.hittable(app.buttons, routine).tap()
        XCTAssertTrue(picker.waitForNonExistence(timeout: 5), "Picking \(routine) must close the routine picker")
    }

    /// Today's resume slab discards the demo workout; with no set logged
    /// it may not ask first.
    static func discardWorkoutInProgress(in app: XCUIApplication) {
        UITestNavigation.selectTab("Today", in: app)
        let resume = app.buttons["resumeSlab.resume"]
        XCTAssertTrue(resume.waitForExistence(timeout: 10), "The demo seed must leave a workout in progress")
        app.buttons["resumeSlab.discard"].tap()
        let alert = app.alerts["Discard workout?"]
        if alert.waitForExistence(timeout: 2) {
            alert.buttons["Discard"].tap()
        }
        XCTAssertTrue(resume.waitForNonExistence(timeout: 5), "Discard must remove the workout in progress")
    }

    /// Makes a routine on Program the way NewBlockSavesUITests does: the
    /// name step, SAVE, then the named existing sessions, then Back.
    static func makeRoutine(_ name: String, adding sessions: [String], in app: XCUIApplication) {
        UITestNavigation.selectTab("Program", in: app)
        UITestNavigation.hittable(app.buttons, "+ NEW").tap()
        UITestNavigation.hittable(app.buttons, "+ New routine").tap()
        UITestNavigation.type(name, into: app.textFields["newRoutine.name"])
        UITestNavigation.hittable(app.buttons, "Save", exactLabel: true).tap()
        XCTAssertTrue(app.textFields["newRoutine.name"].waitForNonExistence(timeout: 10), "SAVE must leave the name step")
        let picker = app.staticTexts.labeled("Existing sessions")
        for session in sessions {
            UITestNavigation.hittable(app.buttons, "+ Add session").tap()
            XCTAssertTrue(picker.waitForExistence(timeout: 10), "+ Add session must list the existing sessions")
            UITestNavigation.hittable(app.buttons, session).tap()
            XCTAssertTrue(picker.waitForNonExistence(timeout: 5), "Adding \(session) must close the session picker")
        }
        UITestNavigation.hittable(app.buttons, "Back").tap()
        let row = app.buttons.matching(NSPredicate(format: "label BEGINSWITH %@", name + ",")).firstMatch
        XCTAssertTrue(row.waitForExistence(timeout: 10), "Program must list \(name)")
    }
}

@MainActor
enum TourProgressLibrary {
    /// The first button whose label begins with `prefix`, such as a row
    /// that reads its title and sub-line as one label.
    static func row(_ prefix: String, in app: XCUIApplication) -> XCUIElement {
        app.buttons.matching(NSPredicate(format: "label BEGINSWITH[c] %@", prefix)).firstMatch
    }

    /// Any element whose label is exactly `label`.
    static func label(_ label: String, in app: XCUIApplication) -> XCUIElement {
        app.descendants(matching: .any).matching(NSPredicate(format: "label == %@", label)).firstMatch
    }

    static func openRow(_ prefix: String, in app: XCUIApplication) {
        tapRevealed(row(prefix, in: app), in: app)
    }

    static func tapRevealed(_ element: XCUIElement, in app: XCUIApplication) {
        XCTAssertTrue(element.waitForExistence(timeout: 10), "Missing \(element)")
        DesignTour.reveal(element, in: app)
        element.tap()
    }

    /// Taps a form row once it sits below `edge`, the lowest control of a
    /// header bar drawn as a top safe-area inset. Rows scroll under such a
    /// bar and XCUITest still calls them hittable, so a plain tap can hit
    /// the bar (Pro Max AX3, New exercise: the equipment list never opened).
    static func tapBelow(_ edge: XCUIElement, _ element: XCUIElement, in app: XCUIApplication) {
        XCTAssertTrue(element.waitForExistence(timeout: 10), "Missing \(element)")
        DesignTour.reveal(element, in: app)
        var swipes = 0
        while element.frame.minY < edge.frame.maxY, swipes < 6 {
            let scroll = app.scrollViews.allElementsBoundByIndex.last(where: { $0.isHittable }) ?? app
            scroll.swipeDown(velocity: .slow)
            swipes += 1
        }
        XCTAssertGreaterThanOrEqual(element.frame.minY, edge.frame.maxY, "\(element) is under the header bar")
        element.tap()
    }

    /// Taps a chip and waits until it reports the selected trait, or the
    /// value "selected", which ProgressFilterChip's tree reads in capitals.
    static func select(_ chip: XCUIElement, in app: XCUIApplication) {
        XCTAssertTrue(chip.waitForExistence(timeout: 10), "Missing \(chip)")
        chip.tap()
        let selected = XCTNSPredicateExpectation(
            predicate: NSPredicate(format: "isSelected == true OR value == %@ OR value == %@", "selected", "SELECTED"),
            object: chip
        )
        XCTAssertEqual(XCTWaiter().wait(for: [selected], timeout: 5), .completed, "\(chip) is not selected after a tap")
    }

    /// Drags a sideways chip row at `element`'s height until the element
    /// sits wholly on screen, as MilestoneOneJourneyUITests does.
    static func revealSideways(_ element: XCUIElement, in app: XCUIApplication) {
        XCTAssertTrue(element.waitForExistence(timeout: 10), "Missing \(element)")
        let width = app.frame.width
        // Read the frame before isHittable: an element off the screen's
        // edge has no activation point, and isHittable then fails the test.
        func onScreen() -> Bool {
            let frame = element.frame
            return frame.minX >= 0 && frame.maxX <= width && element.isHittable
        }
        for _ in 0..<12 {
            if onScreen() { return }
            let frame = element.frame
            let moveLeft = frame.midX > width / 2
            let origin = app.coordinate(withNormalizedOffset: .zero)
            let start = origin.withOffset(CGVector(dx: width * (moveLeft ? 0.75 : 0.25), dy: frame.midY))
            let end = origin.withOffset(CGVector(dx: width * (moveLeft ? 0.4 : 0.6), dy: frame.midY))
            start.press(forDuration: 0.1, thenDragTo: end, withVelocity: .slow, thenHoldForDuration: 0.2)
        }
        XCTAssertTrue(onScreen(), "\(element) must be reachable by scrolling sideways")
    }

    /// A Library filter-bar chip, scrolled into view sideways.
    static func tapFilter(_ identifier: String, in app: XCUIApplication) {
        let chip = app.buttons[identifier]
        revealSideways(chip, in: app)
        chip.tap()
    }

    /// Opens a Library row by its name, then waits for the detail.
    static func openExercise(beginningWith name: String, in app: XCUIApplication) {
        let row = app.staticTexts.matching(NSPredicate(format: "label BEGINSWITH %@", name)).firstMatch
        tapRevealed(row, in: app)
        XCTAssertTrue(app.staticTexts.labeled("Exercise").waitForExistence(timeout: 10), "\(name)'s detail never opened")
    }

    /// Exercise detail > VIEW ALL HISTORY.
    static func openExerciseHistory(in app: XCUIApplication) {
        tapRevealed(row("View all history", in: app), in: app)
        XCTAssertTrue(app.staticTexts.labeled("History").waitForExistence(timeout: 10), "VIEW ALL HISTORY never opened")
    }

    /// Progress > Weekly Tonnage, as WeeklyTonnageUITests does.
    static func openWeeklyTonnage(in app: XCUIApplication) {
        openRow("Weekly Tonnage", in: app)
        XCTAssertTrue(app.buttons["Last 8 weeks"].waitForExistence(timeout: 10), "Weekly Tonnage did not open")
        XCTAssertTrue(app.otherElements["weeklyTonnage.chart"].waitForExistence(timeout: 10), "The chart never appeared")
    }

    /// Activates the chart's newest week element, as VoiceOver's double tap
    /// does in WeeklyTonnageUITests, and waits for its detail.
    static func openNewestWeek(in app: XCUIApplication) {
        let chart = app.otherElements["weeklyTonnage.chart"]
        let weeks = chart.descendants(matching: .any).matching(NSPredicate(format: "label BEGINSWITH %@", "Week of "))
        XCTAssertTrue(weeks.firstMatch.waitForExistence(timeout: 10), "The chart reads no week")
        guard let newest = weeks.allElementsBoundByIndex.max(by: { $0.frame.midX < $1.frame.midX }) else { return }
        UITestNavigation.reveal(newest, in: app)
        newest.tap()
        let title = app.staticTexts.matching(NSPredicate(format: "label BEGINSWITH %@", "Week of ")).firstMatch
        XCTAssertTrue(title.waitForExistence(timeout: 10), "The newest week never opened")
    }
}

    /// The demo workout from the resume slab through Finish, then the
    /// finished session's detail, a second start, and a relaunch.
    @MainActor
    func testW03aDemoLogger() throws {
        let app = DesignTour.launch(["-fitbod-seed-demo-workout"], ready: { $0.buttons["resumeSlab.resume"] })
        DesignTour.step("today--resume") {
            DesignTour.shot(app, "today--resume", still: app.buttons["resumeSlab.resume"])
        }
        DesignTour.step("logger--previous") {
            TourLogger.openWorkout(in: app)
            let previous = TourLogger.element(app, beginning: "Previous:")
            TourLogger.revealLoaded(previous, in: app)
            DesignTour.shot(app, "logger--previous")
        }
        let skip = app.buttons.labeled("Skip all warm-up sets")
        DesignTour.step("logger--warmups-skipped") {
            TourLogger.revealLoaded(skip, in: app, up: true)
            skip.tap()
            let skipped = DesignTour.text(app, "Warm-ups skipped")
            DesignTour.expect(skipped, "Warm-ups skipped")
            DesignTour.shot(app, "logger--warmups-skipped")
        }
        DesignTour.step("logger--warmups-undo") {
            let undo = app.buttons.labeled("Undo skipped warm-ups")
            DesignTour.expect(undo, "UNDO")
            UITestNavigation.revealInLogger(undo, in: app)
            undo.tap()
            DesignTour.expect(skip, "The ramp after UNDO")
            DesignTour.shot(app, "logger--warmups-undo")
        }
        DesignTour.step("logger--set-type-menu") {
            let label = app.buttons["set.type.1.S1"]
            TourLogger.revealLoaded(label, in: app)
            label.tap()
            // "Rest-Pause" exists only while the menu is open.
            let marker = app.buttons["Rest-Pause"]
            DesignTour.expect(marker, "The set-type menu")
            DesignTour.shot(app, "logger--set-type-menu")
            DesignTour.dismissDialog(app)
            _ = marker.waitForNonExistence(timeout: 5)
        }
        DesignTour.step("logger--plate-stack") {
            // Each warm-up row already draws a plate stack read as "Bar: …",
            // so wait for one more.
            let stacks = app.descendants(matching: .any).matching(NSPredicate(format: "label BEGINSWITH %@", "Bar:"))
            let weight = app.textFields.matching(identifier: "set.weight.1.S1").firstMatch
            TourLogger.revealLoaded(weight, in: app)
            let before = stacks.count
            // SetRow toggles the disclosure on a tap in the weight cell, but
            // the cell has its own tap gesture over the same area. Try the
            // cell's padding above and below the field, then the field.
            let spots: [(name: String, offset: CGVector, nudge: CGFloat)] = [
                ("the padding above the field", CGVector(dx: 0.5, dy: 0), -6),
                ("the padding below the field", CGVector(dx: 0.5, dy: 1), 6),
                ("the field", CGVector(dx: 0.5, dy: 0.5), 0),
            ]
            var opened: String?
            for spot in spots where opened == nil {
                weight.coordinate(withNormalizedOffset: spot.offset).withOffset(CGVector(dx: 0, dy: spot.nudge)).tap()
                if app.keyboards.firstMatch.waitForExistence(timeout: 1) {
                    UITestNavigation.hittable(app.buttons, "Done", exactLabel: true).tap()
                    _ = app.keyboards.firstMatch.waitForNonExistence(timeout: 5)
                }
                let deadline = Date.now.addingTimeInterval(3)
                while stacks.count <= before && Date.now < deadline { Thread.sleep(forTimeInterval: 0.25) }
                if stacks.count > before { opened = spot.name }
            }
            if let opened {
                XCTContext.runActivity(named: "The plate stack opened on a tap in \(opened)") { _ in }
                DesignTour.shot(app, "logger--plate-stack")
            } else {
                // An app problem, recorded in the audit: the cell's gesture
                // wins, so the row's toggle never runs.
                XCTContext.runActivity(named: "No tap in set 1's weight cell opened its plate stack (tried \(spots.map(\.name).joined(separator: ", ")))") { _ in }
            }
        }
        DesignTour.step("logger--pr-banner") {
            // The seeded history tops out at 5 reps, so 9 is a record
            // (WorkoutLoggerJourneyUITests).
            TourLogger.logSet("1.S1", reps: "9", in: app)
            let banner = DesignTour.any(app, containing: "personal record")
            DesignTour.expect(banner, "The PR banner", timeout: 2)
            DesignTour.shot(app, "logger--pr-banner")
        }
        let status = TourLogger.restStatus(in: app)
        DesignTour.step("logger--rest-resting") {
            DesignTour.expect(status, "The rest dock")
            DesignTour.expect(DesignTour.text(app, containing: "RESTING"), "RESTING")
            DesignTour.shot(app, "logger--rest-resting", still: status)
        }
        DesignTour.step("logger--rest-complete") {
            // The bench press rests 180 s: twelve taps of -15 s end it.
            let minus = app.buttons["rest.minus15"]
            for _ in 0..<16 where status.exists && !status.label.hasSuffix("rest complete") {
                minus.tap()
            }
            XCTAssertTrue(status.label.hasSuffix("rest complete"), "The dock must read REST COMPLETE: \(status.label)")
            DesignTour.shot(app, "logger--rest-complete", still: status)
        }
        DesignTour.step("workout-summary--after-finish") {
            DesignTour.tap(app.buttons, "More", exact: true)
            DesignTour.tap(app.buttons, "Finish", exact: true)
            DesignTour.waitForDialog(app, title: "Finish workout?")
            DesignTour.tapDialogButton(app, "Finish")
            let done = app.buttons["summary.done"]
            DesignTour.expect(done, "The summary after Finish")
            DesignTour.expect(app.staticTexts["WORKOUT COMPLETE"], "WORKOUT COMPLETE")
            DesignTour.shot(app, "workout-summary--after-finish", still: app.staticTexts["WORKOUT COMPLETE"])
            done.tap()
            DesignTour.expect(app.buttons["today.sessions"], "Today after the summary")
        }
        // The demo's Push A is in no routine, so Today's picker never lists
        // it; W03c starts a second workout from an ended block's session.
    }

    // MARK: W01 fresh

    @MainActor
    func testW01Fresh() throws {
        // DesignTour.launch settles the install; the second, hand-made reset
        // launch is shot before the tab bar appears.
        let app = DesignTour.launch()
        app.terminate()
        app.launchArguments = [UITestNavigation.resetFlag, UITestNavigation.resetSignalFlag]
        app.launch()
        DesignTour.step("preparing-library") {
            let splash = DesignTour.any(app, containing: "Preparing library")
            // launch() returns once the app is idle, which is often after the
            // import, so a miss is no failure: the map then takes it by hand.
            if splash.waitForExistence(timeout: 10) {
                DesignTour.shot(app, "preparing-library", still: splash)
            } else {
                XCTContext.runActivity(named: "The Preparing library splash was gone before the shot") { _ in }
            }
        }
        XCTAssertTrue(UITestNavigation.tab("Today", in: app).waitForExistence(timeout: 90), "The tab bar never appeared")
        UITestNavigation.assertFreshStore(in: app)

        UITestNavigation.selectTab("Progress", in: app)
        DesignTour.step("progress--empty") {
            DesignTour.expect(DesignTour.text(app, "No history yet"), "Progress's empty Recent")
            DesignTour.shot(app, "progress--empty")
        }
        DesignTour.step("history--empty") {
            app.buttons["progress.history"].tap()
            DesignTour.expect(DesignTour.text(app, "No sessions yet"), "History's empty state")
            DesignTour.shot(app, "history--empty")
            DesignTour.tap(app.buttons, "Back", exact: true)
        }

        UITestNavigation.selectTab("Library", in: app)
        let search = app.textFields["library.search.browser"]
        DesignTour.expect(search, "Library")
        let facets: [(id: String, shot: String, pick: String?)] = [
            ("library.filter.muscle", "filter-picker--muscle", "Chest"),
            ("library.filter.equipment", "filter-picker--equipment", "Barbell"),
            // Mechanic stays unpicked, so the list keeps every match.
            ("library.filter.mechanic", "filter-picker--mechanic", nil),
        ]
        for facet in facets {
            DesignTour.step(facet.shot) {
                TourProgressLibrary.tapFilter(facet.id, in: app)
                let done = app.buttons.labeled("Done")
                DesignTour.expect(done, "The \(facet.shot) sheet")
                if let pick = facet.pick {
                    DesignTour.tap(app.buttons, pick, exact: true)
                }
                DesignTour.shot(app, facet.shot, still: done)
                DesignTour.tap(app.buttons, "Done", exact: true)
                _ = done.waitForNonExistence(timeout: 5)
            }
        }
        DesignTour.step("library--filtered") {
            let bench = app.staticTexts.matching(NSPredicate(format: "label BEGINSWITH %@", "Barbell Bench Press - Medium Grip")).firstMatch
            DesignTour.expect(bench, "Chest and barbell matches")
            DesignTour.shot(app, "library--filtered")
        }
        DesignTour.step("exercise-detail--not-logged") {
            TourProgressLibrary.openExercise(beginningWith: "Barbell Bench Press - Medium Grip", in: app)
            DesignTour.expect(DesignTour.text(app, containing: "Not logged yet"), "Recent performance's empty line")
            DesignTour.shot(app, "exercise-detail--not-logged")
        }
        DesignTour.step("exercise-history--empty") {
            TourProgressLibrary.openExerciseHistory(in: app)
            DesignTour.expect(DesignTour.text(app, "No logged sets yet"), "The empty exercise history")
            DesignTour.shot(app, "exercise-history--empty")
            DesignTour.tap(app.buttons, "Back", exact: true)
            DesignTour.expect(app.staticTexts.labeled("Exercise"), "Exercise detail after its history")
            DesignTour.tap(app.buttons, "Back", exact: true)
            DesignTour.expect(search, "Library after the detail")
        }
        let probe = "Zercher Probe"
        DesignTour.step("library--no-matches") {
            TourProgressLibrary.tapFilter("library.clearFilters", in: app)
            UITestNavigation.type(probe, into: search, replacing: false)
            DesignTour.expect(DesignTour.text(app, "No exercises match \"\(probe)\""), "The no-match state")
            DesignTour.shot(app, "library--no-matches")
        }
        let name = app.textFields["customExercise.name"]
        DesignTour.step("custom-exercise--new") {
            DesignTour.tap(app.buttons, "Create Custom Exercise", exact: true)
            DesignTour.expect(name, "The editor from the no-match state")
            XCTAssertEqual(name.value as? String, probe, "The name must start as the search text")
            DesignTour.shot(app, "custom-exercise--new")
        }
        DesignTour.step("custom-exercise--validation") {
            // No name and no primary muscle: SAVE shows both errors.
            UITestNavigation.type("", into: name)
            DesignTour.tap(app.buttons, "Save", exact: true)
            DesignTour.expect(app.staticTexts["customExercise.nameError"], "The name error")
            DesignTour.shot(app, "custom-exercise--validation")
        }
        DesignTour.step("exercise-detail--custom") {
            UITestNavigation.type(probe, into: name)
            TourProgressLibrary.tapRevealed(app.buttons.labeled("Add Primary Muscle"), in: app)
            let chest = app.buttons["musclePicker.chest"]
            DesignTour.expect(chest, "The muscle picker")
            chest.tap()
            DesignTour.tap(app.buttons, "Save", exact: true)
            XCTAssertTrue(name.waitForNonExistence(timeout: 5), "SAVE must close the editor")
            // The search still reads the new name.
            let row = app.staticTexts[probe]
            DesignTour.expect(row, "The saved custom exercise")
            row.tap()
            DesignTour.expect(app.buttons.labeled("Edit"), "The custom exercise's detail")
            DesignTour.shot(app, "exercise-detail--custom")
        }
        DesignTour.step("custom-exercise--edit") {
            DesignTour.tap(app.buttons, "Edit", exact: true)
            DesignTour.expect(app.staticTexts.labeled("Edit exercise"), "Edit exercise")
            DesignTour.shot(app, "custom-exercise--edit")
        }
        // Last: the system photo picker; the walk ends with terminate.
        DesignTour.step("photos-picker") {
            let add = app.buttons.labeled("Add photo")
            TourProgressLibrary.tapRevealed(add, in: app)
            let covered = XCTNSPredicateExpectation(predicate: NSPredicate(format: "hittable == false"), object: add)
            _ = XCTWaiter().wait(for: [covered], timeout: 5)
            DesignTour.shot(app, "photos-picker")
        }
        app.terminate()
    }
